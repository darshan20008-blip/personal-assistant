        let scene, camera, renderer, sphere, particlesMesh, audioCtx, analyser, microphone, dataArray;
        let isListening = false;
        let currentStyle = 'wireframe';
        let clock = new THREE.Clock();

        function init3D() {
            const container = document.getElementById('canvas-container');
            
            scene = new THREE.Scene();
            camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
            camera.position.z = 5;

            renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
            container.appendChild(renderer.domElement);

            // Lighting
            const ambientLight = new THREE.AmbientLight(0xffffff, 0.8);
            scene.add(ambientLight);

            const pointLight = new THREE.PointLight(0x00ffff, 2, 50);
            pointLight.position.set(5, 5, 5);
            scene.add(pointLight);

            const pointLight2 = new THREE.PointLight(0x8b5cf6, 2, 50);
            pointLight2.position.set(-5, -5, -5);
            scene.add(pointLight2);

            buildSphere(currentStyle);

            window.addEventListener('resize', onWindowResize);
            document.getElementById('code-content').textContent = document.documentElement.outerHTML;
            animate();
        }

        function buildSphere(style) {
            if (sphere) scene.remove(sphere);
            if (particlesMesh) scene.remove(particlesMesh);

            if (style === 'wireframe') {
                const geometry = new THREE.IcosahedronGeometry(1.8, 4);
                const material = new THREE.MeshStandardMaterial({
                    color: 0x00ffff,
                    wireframe: true,
                    roughness: 0.1,
                    metalness: 0.8
                });
                sphere = new THREE.Mesh(geometry, material);
                scene.add(sphere);
            } else if (style === 'normal') {
                const geometry = new THREE.SphereGeometry(1.6, 64, 64);
                const material = new THREE.MeshNormalMaterial({ wireframe: false });
                sphere = new THREE.Mesh(geometry, material);
                scene.add(sphere);
            } else if (style === 'crystal') {
                const geometry = new THREE.DodecahedronGeometry(1.7, 1);
                const material = new THREE.MeshPhysicalMaterial({
                    color: 0x3b82f6,
                    roughness: 0.2,
                    metalness: 0.5,
                    transmission: 0.6,
                    thickness: 1.2,
                    wireframe: false
                });
                sphere = new THREE.Mesh(geometry, material);
                scene.add(sphere);
            } else if (style === 'particles') {
                const particleCount = 2500;
                const geometry = new THREE.BufferGeometry();
                const positions = new Float32Array(particleCount * 3);

                for (let i = 0; i < particleCount * 3; i += 3) {
                    const u = Math.random();
                    const v = Math.random();
                    const theta = u * 2.0 * Math.PI;
                    const phi = Math.acos(2.0 * v - 1.0);
                    const r = 1.6 + (Math.random() - 0.5) * 0.3;
                    
                    positions[i] = r * Math.sin(phi) * Math.cos(theta);
                    positions[i+1] = r * Math.sin(phi) * Math.sin(theta);
                    positions[i+2] = r * Math.cos(phi);
                }

                geometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));
                const material = new THREE.PointsMaterial({
                    size: 0.04,
                    color: 0x06b6d4,
                    transparent: true,
                    opacity: 0.8
                });
                particlesMesh = new THREE.Points(geometry, material);
                scene.add(particlesMesh);
            }
        }

        function updateSphereStyle() {
            currentStyle = document.getElementById('sphere-style').value;
            buildSphere(currentStyle);
        }

        async function toggleMicrophone() {
            const micBtn = document.getElementById('mic-btn');
            const micIcon = document.getElementById('mic-icon');
            const micText = document.getElementById('mic-text');
            const statusDot = document.getElementById('status-dot');
            const alertBox = document.getElementById('alert-box');

            if (!isListening) {
                try {
                    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
                    const stream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false });
                    
                    analyser = audioCtx.createAnalyser();
                    analyser.fftSize = 256;
                    
                    microphone = audioCtx.createMediaStreamSource(stream);
                    microphone.connect(analyser);
                    
                    dataArray = new Uint8Array(analyser.frequencyBinCount);
                    
                    isListening = true;
                    micBtn.classList.remove('from-cyan-500', 'to-blue-600', 'shadow-cyan-500/25');
                    micBtn.classList.add('from-red-500', 'to-rose-600', 'shadow-rose-500/25');
                    micIcon.className = 'fa-solid fa-microphone-slash text-lg';
                    micText.textContent = 'Stop Voice Input';
                    statusDot.className = 'w-3 h-3 rounded-full bg-rose-500 animate-ping';
                    alertBox.classList.add('hidden');
                } catch (err) {
                    console.error('Microphone error:', err);
                    alertBox.classList.remove('hidden');
                    document.getElementById('alert-text').textContent = 'Microphone permission denied or unavailable.';
                }
            } else {
                if (audioCtx) audioCtx.close();
                isListening = false;
                micBtn.classList.remove('from-red-500', 'to-rose-600', 'shadow-rose-500/25');
                micBtn.classList.add('from-cyan-500', 'to-blue-600', 'shadow-cyan-500/25');
                micIcon.className = 'fa-solid fa-microphone text-lg';
                micText.textContent = 'Start Voice Input';
                statusDot.className = 'w-3 h-3 rounded-full bg-cyan-400 animate-pulse';
                document.getElementById('volume-bar').style.width = '0%';
            }
        }

        function animate() {
            requestAnimationFrame(animate);

            const delta = clock.getDelta();
            let voiceVolume = 0;

            if (isListening && analyser) {
                analyser.getByteFrequencyData(dataArray);
                let sum = 0;
                for (let i = 0; i < dataArray.length; i++) {
                    sum += dataArray[i];
                }
                voiceVolume = (sum / dataArray.length) / 255.0; // Normalized 0 to 1
                document.getElementById('volume-bar').style.width = `${Math.min(voiceVolume * 150, 100)}%`;
            }

            const sensitivity = parseFloat(document.getElementById('sensitivity-slider').value);
            const dynamicSpeed = 0.5 + (voiceVolume * sensitivity * 8.0);

            // Rotate sphere automatically and faster when speaking
            const targetMesh = sphere || particlesMesh;
            if (targetMesh) {
                targetMesh.rotation.x += delta * dynamicSpeed * 0.6;
                targetMesh.rotation.y += delta * dynamicSpeed * 0.9;

                // Pulse scale based on voice intensity
                const targetScale = 1.0 + (voiceVolume * sensitivity * 0.6);
                targetMesh.scale.lerp(new THREE.Vector3(targetScale, targetScale, targetScale), 0.2);
            }

            renderer.render(scene, camera);
        }

        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        function toggleModal(id) {
            const modal = document.getElementById(id);
            modal.classList.toggle('hidden');
        }

        function copyCode() {
            const codeText = document.documentElement.outerHTML;
            const textarea = document.createElement('textarea');
            textarea.value = codeText;
            document.body.appendChild(textarea);
            textarea.select();
            try {
                document.execCommand('copy');
                alert('Code copied to clipboard successfully!');
            } catch (err) {
                console.error('Failed to copy code', err);
            }
            document.body.removeChild(textarea);
        }

        window.onload = init3D;
    </script>
